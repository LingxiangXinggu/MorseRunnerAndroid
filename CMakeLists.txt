cmake_minimum_required(VERSION 3.22)
project(MorseRunnerAndroid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mr-core STATIC core/engine.cpp)
target_include_directories(mr-core PUBLIC core)
set_target_properties(mr-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  add_library(morserunner SHARED app/src/main/cpp/bridge.cpp)
  target_link_libraries(morserunner PRIVATE mr-core)
  target_link_options(morserunner PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(core_test tests/core_test.cpp)
  target_link_libraries(core_test PRIVATE mr-core)
  enable_testing()
  add_test(NAME core_test COMMAND core_test)
endif()
